# 流程示例

## 单线撰写

制定大纲后引用容器「成稿」写出全文，再引用库节点「校对全文」校对，定稿写入工作区 `out/单线定稿.md`。

```json
{
  "Nodes": [
    {
      "Name": "撰写正文",
      "Kind": "Llm",
      "Prompt": "按大纲撰写正文。",
      "Tools": ["file"],
      "ToolPorts": { "file": { "root": "TASK_WORK_PATH" } },
      "Inputs": [
        { "Name": "大纲", "Description": "待撰写的章节大纲。" },
        { "Name": "TASK_WORK_PATH", "Type": "Text", "Static": true, "Description": "本次任务目录，文件工具的路径基准。" }
      ],
      "Outputs": [{ "Name": "正文", "Type": "Text", "Description": "按大纲写好的章节正文。" }]
    },
    {
      "Name": "合成全文",
      "Kind": "Llm",
      "Prompt": "把正文合成一篇完整文档。",
      "Inputs": [{ "Name": "正文", "Description": "章节正文。" }],
      "Outputs": [{ "Name": "全文", "Type": "Text" }]
    },
    {
      "Name": "校对全文",
      "Kind": "Llm",
      "Prompt": "校对全文，交回修正后的定稿。",
      "Inputs": [{ "Name": "全文", "Description": "待校对的全文。" }],
      "Outputs": [{ "Name": "定稿", "Type": "Text" }]
    },
    {
      "Name": "成稿",
      "Kind": "Container",
      "Inputs": [
        { "Name": "大纲", "Description": "待撰写的章节大纲。" },
        { "Name": "TASK_WORK_PATH", "Type": "Text", "Static": true, "Description": "本次任务目录。" },
        { "Name": "MODEL_NAME", "Type": "Text", "Static": true, "Description": "执行使用的模型，由宿主给出。" }
      ],
      "Outputs": [{ "Name": "全文", "Ref": "合成@全文", "Description": "容器内写好的全文。" }],
      "Nodes": [
        { "Label": "撰写", "Use": "撰写正文", "Static": { "MODEL_NAME": "@MODEL_NAME", "TASK_WORK_PATH": "@TASK_WORK_PATH" }, "From": [{ "大纲": "@大纲" }] },
        { "Label": "合成", "Use": "合成全文", "Static": { "MODEL_NAME": "@MODEL_NAME" }, "From": [{ "正文": "撰写@正文" }] }
      ]
    }
  ],
  "Flows": [
    {
      "Name": "单线撰写",
      "Description": "制定大纲后经容器写出全文，再引用库节点校对，定稿写入文件",
      "Start": {
        "Outputs": [
          { "Name": "目标", "Type": "Text", "Description": "要写的文档主题。" },
          { "Name": "风格", "Type": "Text", "Description": "写作风格要求。" },
          { "Name": "WORKSPACE_PATH", "Type": "Text", "Static": true, "Description": "工作区根目录。" },
          { "Name": "TASK_WORK_PATH", "Type": "Text", "Static": true, "Description": "本次任务目录。" },
          { "Name": "MODEL_NAME", "Type": "Text", "Static": true }
        ]
      },
      "HostStartInputs": {
        "WORKSPACE_PATH": "WORKSPACE_PATH",
        "TASK_WORK_PATH": "TASK_WORK_PATH",
        "MODEL_NAME": "MODEL"
      },
      "Nodes": [
        {
          "Name": "整体",
          "Kind": "Container",
          "Inputs": [
            { "Name": "目标" },
            { "Name": "风格" },
            { "Name": "WORKSPACE_PATH", "Type": "Text", "Static": true },
            { "Name": "TASK_WORK_PATH", "Type": "Text", "Static": true },
            { "Name": "MODEL_NAME", "Type": "Text", "Static": true }
          ],
          "In": {
            "目标": "Start@目标",
            "风格": "Start@风格",
            "WORKSPACE_PATH": "Start@WORKSPACE_PATH",
            "TASK_WORK_PATH": "Start@TASK_WORK_PATH",
            "MODEL_NAME": "Start@MODEL_NAME"
          },
          "Nodes": [
            {
              "Name": "制定大纲",
              "Kind": "Llm",
              "Static": { "MODEL_NAME": "@MODEL_NAME" },
              "Prompt": "写出可撰写的章节大纲。",
              "Inputs": [
                { "Name": "目标", "Description": "任务目标。" },
                { "Name": "风格", "Description": "写作风格要求。" }
              ],
              "Outputs": [{ "Name": "大纲", "Type": "Text" }],
              "From": [{ "目标": "@目标", "风格": "@风格" }]
            },
            {
              "Label": "成稿",
              "Use": "成稿",
              "In": {
                "大纲": "制定大纲@大纲",
                "TASK_WORK_PATH": "@TASK_WORK_PATH",
                "MODEL_NAME": "@MODEL_NAME"
              }
            },
            {
              "Label": "校对",
              "Use": "校对全文",
              "Static": { "MODEL_NAME": "@MODEL_NAME" },
              "From": [{ "全文": "成稿@全文" }]
            },
            {
              "Name": "保存定稿",
              "Kind": "FileWrite",
              "Path": "{root}/out/单线定稿.md",
              "Variables": { "root": "WORKSPACE_PATH" },
              "Inputs": [{ "Name": "正文" }, { "Name": "WORKSPACE_PATH", "Type": "Text", "Static": true }],
              "Static": { "WORKSPACE_PATH": "@WORKSPACE_PATH" },
              "From": [{ "正文": "校对@定稿" }]
            }
          ]
        }
      ]
    }
  ]
}
```

## 取用任务工作路径

开始节点声明 `目标`、`WORKSPACE_PATH` 与 `TASK_WORK_PATH`，后两个端口是静态端口，`HostStartInputs` 把它们绑定到同名宿主输入，把模型端口绑定到 `MODEL`，由任务启动时给值。容器把这些静态端口逐层转发，执行节点用 `Static` 绑定取用。文件写入节点用 `Variables` 把 `root` 变量映射到 `TASK_WORK_PATH` 端口，`Path` 写成 `{root}/result.md`，因此落在本次任务目录下。

两个路径端口写在 `Start.Outputs` 里，提交任务时作为只读项展示，不填写。`整体` 把三个开始端口接进自己的传入端口，成员经 `@端口` 取用，成员不直接引用开始节点。`ToolPorts` 把文件工具的 `root` 变量绑到 `TASK_WORK_PATH` 端口，工具因此以本次任务目录为路径基准。

```json
{
  "Flows": [
    {
      "Name": "落盘位置",
      "Start": {
        "Outputs": [
          { "Name": "目标", "Type": "Text", "Description": "要完成的目标。" },
          { "Name": "WORKSPACE_PATH", "Type": "Text", "Static": true },
          { "Name": "TASK_WORK_PATH", "Type": "Text", "Static": true },
          { "Name": "MODEL_NAME", "Type": "Text", "Static": true }
        ]
      },
      "HostStartInputs": {
        "WORKSPACE_PATH": "WORKSPACE_PATH",
        "TASK_WORK_PATH": "TASK_WORK_PATH",
        "MODEL_NAME": "MODEL"
      },
      "Nodes": [
        {
          "Name": "整体",
          "Kind": "Container",
          "Inputs": [
            { "Name": "目标" },
            { "Name": "WORKSPACE_PATH", "Type": "Text", "Static": true },
            { "Name": "TASK_WORK_PATH", "Type": "Text", "Static": true },
            { "Name": "MODEL_NAME", "Type": "Text", "Static": true }
          ],
          "In": {
            "目标": "Start@目标",
            "WORKSPACE_PATH": "Start@WORKSPACE_PATH",
            "TASK_WORK_PATH": "Start@TASK_WORK_PATH",
            "MODEL_NAME": "Start@MODEL_NAME"
          },
          "Nodes": [
            {
              "Name": "实施",
              "Kind": "Llm",
              "Static": { "MODEL_NAME": "@MODEL_NAME", "WORKSPACE_PATH": "@WORKSPACE_PATH", "TASK_WORK_PATH": "@TASK_WORK_PATH" },
              "Prompt": "在本次任务目录下完成目标。",
              "Tools": ["file"],
              "ToolPorts": { "file": { "root": "TASK_WORK_PATH" } },
              "Inputs": [
                { "Name": "目标" },
                { "Name": "WORKSPACE_PATH", "Type": "Text", "Static": true },
                { "Name": "TASK_WORK_PATH", "Type": "Text", "Static": true }
              ],
              "Outputs": [{ "Name": "结果", "Type": "Text" }],
              "From": [
                {
                  "目标": "@目标"
                }
              ]
            },
            {
              "Name": "保存结果",
              "Kind": "FileWrite",
              "Path": "{root}/result.md",
              "Variables": { "root": "TASK_WORK_PATH" },
              "Inputs": [{ "Name": "正文" }, { "Name": "TASK_WORK_PATH", "Type": "Text", "Static": true }],
              "Static": { "TASK_WORK_PATH": "@TASK_WORK_PATH" },
              "From": [{ "正文": "实施@结果" }]
            }
          ]
        }
      ]
    }
  ]
}
```
