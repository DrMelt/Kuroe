# 流程示例

## 单线撰写

制定大纲后引用容器「成稿」写出全文，再引用库节点「校对全文」校对，定稿写入 `out/单线定稿.md`。

示例覆盖节点库定义、容器模型槽位声明、容器引用、库节点引用、文件写入节点与 `BasePath` 基准输入。

```json
{
  "Nodes": [
    {
      "Name": "撰写正文",
      "Kind": "Llm",
      "Prompt": "按大纲撰写正文。",
      "Tools": ["file"],
      "Inputs": [{ "Name": "大纲", "Description": "待撰写的章节大纲。" }],
      "Outputs": [{ "Name": "正文", "Type": "Text", "Description": "按大纲写好的章节正文。" }]
    },
    {
      "Name": "合成全文",
      "Kind": "Llm",
      "Prompt": "把正文合成一篇完整文档。",
      "Inputs": [{ "Name": "正文", "Description": "章节正文。" }],
      "Outputs": [{ "Name": "全文", "Type": "Text" }]
    },
    {
      "Name": "校对全文",
      "Kind": "Llm",
      "Prompt": "校对全文，交回修正后的定稿。",
      "Inputs": [{ "Name": "全文", "Description": "待校对的全文。" }],
      "Outputs": [{ "Name": "定稿", "Type": "Text" }]
    },
    {
      "Name": "成稿",
      "Kind": "Container",
      "Models": ["章节作者", "统稿者"],
      "Inputs": [{ "Name": "大纲", "Description": "待撰写的章节大纲。" }],
      "Outputs": [{ "Name": "全文", "Ref": "合成@全文", "Description": "容器内写好的全文。" }],
      "Nodes": [
        { "Label": "撰写", "Use": "撰写正文", "Model": "章节作者", "From": [{ "大纲": "@大纲" }] },
        { "Label": "合成", "Use": "合成全文", "Model": "统稿者", "From": [{ "正文": "撰写@正文" }] }
      ]
    }
  ],
  "Flows": [
    {
      "Name": "单线撰写",
      "Description": "制定大纲后经容器写出全文，再引用库节点校对，定稿写入文件",
      "Models": [
        { "Name": "规划者", "Model": "<模型>" },
        { "Name": "章节作者", "Model": "<模型>" },
        { "Name": "统稿者", "Model": "<模型>" },
        { "Name": "校对者", "Model": "<模型>" }
      ],
      "Start": {
        "Outputs": [
          { "Name": "目标", "Type": "Text", "Description": "要写的文档主题。" },
          { "Name": "风格", "Type": "Text", "Description": "写作风格要求。" },
          { "Name": "基准", "Type": "Text", "Value": "." }
        ]
      },
      "Nodes": [
        {
          "Name": "整体",
          "Kind": "Container",
          "Nodes": [
            {
              "Name": "制定大纲",
              "Kind": "Llm",
              "Model": "规划者",
              "Prompt": "写出可撰写的章节大纲。",
              "Inputs": [
                { "Name": "目标", "Description": "任务目标。" },
                { "Name": "风格", "Description": "写作风格要求。" }
              ],
              "Outputs": [{ "Name": "大纲", "Type": "Text" }],
              "From": [{ "目标": "Start@目标", "风格": "Start@风格" }]
            },
            {
              "Label": "成稿",
              "Use": "成稿",
              "Models": { "章节作者": "章节作者", "统稿者": "统稿者" },
              "In": { "大纲": "制定大纲@大纲" }
            },
            {
              "Label": "校对",
              "Use": "校对全文",
              "Model": "校对者",
              "From": [{ "全文": "成稿@全文" }]
            },
            {
              "Name": "保存定稿",
              "Kind": "FileWrite",
              "Path": "out/单线定稿.md",
              "Inputs": [{ "Name": "正文" }, { "Name": "BasePath" }],
              "From": [{ "正文": "校对@定稿", "BasePath": "Start@基准" }]
            }
          ]
        }
      ]
    }
  ]
}
```

## 取用任务工作路径

开始节点声明 `目标`、`启动路径` 与 `流程路径`，`HostStartInputs` 把后两个绑定到宿主输入，由任务启动时给值。执行节点把两个路径接到输入端口，模型据此知道文件该读写在哪一层目录。文件写入节点把 `流程路径` 接到 `BasePath`，相对路径因此落在本次任务目录下。

示例覆盖宿主开始输入、`BasePath` 基准输入与作用域内的 `Start@端口` 引用。两个路径端口写在 `Start.Outputs` 里，提交任务时作为只读项展示，不填写。

```json
{
  "Flows": [
    {
      "Name": "落盘位置",
      "Models": [{ "Name": "执行者", "Model": "<模型>" }],
      "Start": {
        "Outputs": [
          { "Name": "目标", "Type": "Text", "Description": "要完成的目标。" },
          { "Name": "启动路径", "Type": "Text" },
          { "Name": "流程路径", "Type": "Text" }
        ]
      },
      "HostStartInputs": {
        "启动路径": "WORKSPACE_PATH",
        "流程路径": "TASK_WORK_PATH"
      },
      "Nodes": [
        {
          "Name": "整体",
          "Kind": "Container",
          "Nodes": [
            {
              "Name": "实施",
              "Kind": "Llm",
              "Model": "执行者",
              "Prompt": "在流程路径下完成目标。",
              "Tools": ["file"],
              "Inputs": [
                { "Name": "目标" },
                { "Name": "启动路径" },
                { "Name": "流程路径" }
              ],
              "Outputs": [{ "Name": "结果", "Type": "Text" }],
              "From": [
                {
                  "目标": "Start@目标",
                  "启动路径": "Start@启动路径",
                  "流程路径": "Start@流程路径"
                }
              ]
            },
            {
              "Name": "保存结果",
              "Kind": "FileWrite",
              "Path": "result.md",
              "Inputs": [{ "Name": "正文" }, { "Name": "BasePath" }],
              "From": [{ "正文": "实施@结果", "BasePath": "Start@流程路径" }]
            }
          ]
        }
      ]
    }
  ]
}
```
