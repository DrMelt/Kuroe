# 流程示例

## 单线撰写

拟定大纲后引用容器「成稿」写出全文，再引用库节点「校对全文」校对，定稿写入本次任务目录的 `单线定稿.md`。

目标与风格在提交时给一次，声明为常量开始端口。拟定大纲声明 `Emission: Once`，输入全部是常量，静态输入就绪后执行并首次发布后冻结。成稿容器只包含撰写成员，直接对外输出撰写正文，不再经过额外的合成节点。

```json
{
  "Nodes": [
    {
      "Name": "撰写正文",
      "Emission": "Once",
      "Kind": "Llm",
      "Prompt": "按大纲撰写正文。",
      "Inputs": [
        { "Name": "大纲", "Type": "Text", "Mode": "Constant", "Role": "Content", "Description": "待撰写的章节大纲。" }
      ],
      "Outputs": [{ "Name": "正文", "Type": "Text", "Mode": "Constant", "Description": "按大纲写好的章节正文。" }]
    },
    {
      "Name": "拟定大纲",
      "Emission": "Once",
      "Kind": "Llm",
      "Prompt": "写出可撰写的章节大纲。",
      "Inputs": [
        { "Name": "目标", "Type": "Text", "Mode": "Constant", "Role": "Content", "Description": "任务目标。" },
        { "Name": "风格", "Type": "Text", "Mode": "Constant", "Role": "Content", "Description": "写作风格要求。" }
      ],
      "Outputs": [{ "Name": "大纲", "Type": "Text", "Mode": "Constant" }]
    },
    {
      "Name": "校对全文",
      "Kind": "Llm",
      "Emission": "Once",
      "Prompt": "校对全文，交回修正后的定稿。",
      "Inputs": [{ "Name": "全文", "Type": "Text", "Mode": "Constant", "Role": "Content", "Description": "待校对的全文。" }],
      "Outputs": [{ "Name": "定稿", "Type": "Text", "Mode": "Constant" }]
    },
    {
      "Name": "成稿",
      "Kind": "Container",
      "Inputs": [
        { "Name": "大纲", "Type": "Text", "Mode": "Constant", "Role": "Content", "Description": "待撰写的章节大纲。" },
        { "Name": "MODEL_NAME", "Type": "Text", "Mode": "Constant", "Description": "执行使用的模型，由宿主给出。" }
      ],
      "Outputs": [{ "Name": "全文", "Type": "Text", "Mode": "Constant", "Ref": "撰写@正文", "Description": "容器内写好的全文。" }],
      "Nodes": [
        { "Label": "撰写", "Use": "撰写正文", "Bindings": { "大纲": "@大纲", "MODEL_NAME": "@MODEL_NAME" } }
      ]
    }
  ],
  "Flows": [
    {
      "Name": "单线撰写",
      "Description": "拟定大纲后经容器写出全文，再引用库节点校对，定稿写入本次任务目录",
      "Start": {
        "Outputs": [
          { "Name": "目标", "Type": "Text", "Mode": "Constant", "Description": "要写的文档主题。" },
          { "Name": "风格", "Type": "Text", "Mode": "Constant", "Description": "写作风格要求。" },
          { "Name": "TASK_WORK_PATH", "Type": "Text", "Mode": "Constant", "Description": "本次任务目录。" },
          { "Name": "MODEL_NAME", "Type": "Text", "Mode": "Constant" }
        ]
      },
      "HostStartInputs": {
        "TASK_WORK_PATH": "TASK_WORK_PATH",
        "MODEL_NAME": "MODEL"
      },
      "Nodes": [
        {
          "Name": "整体",
          "Kind": "Container",
          "Inputs": [
            { "Name": "目标", "Type": "Text", "Mode": "Constant", "Role": "Content" },
            { "Name": "风格", "Type": "Text", "Mode": "Constant", "Role": "Content" },
            { "Name": "TASK_WORK_PATH", "Type": "Text", "Mode": "Constant" },
            { "Name": "MODEL_NAME", "Type": "Text", "Mode": "Constant" }
          ],
          "Bindings": {
            "目标": "Start@目标",
            "风格": "Start@风格",
            "TASK_WORK_PATH": "Start@TASK_WORK_PATH",
            "MODEL_NAME": "Start@MODEL_NAME"
          },
          "Nodes": [
            {
              "Label": "拟定",
              "Use": "拟定大纲",
              "Bindings": { "目标": "@目标", "风格": "@风格", "MODEL_NAME": "@MODEL_NAME" }
            },
            {
              "Label": "成稿",
              "Use": "成稿",
              "Bindings": {
                "大纲": "拟定@大纲",
                "MODEL_NAME": "@MODEL_NAME"
              }
            },
            {
              "Label": "校对",
              "Use": "校对全文",
              "Bindings": { "全文": "成稿@全文", "MODEL_NAME": "@MODEL_NAME" }
            },
            {
              "Name": "保存定稿",
              "Kind": "FileWrite",
              "Path": "{root}/单线定稿.md",
              "Variables": { "root": "TASK_WORK_PATH" },
              "Inputs": [
                { "Name": "正文", "Type": "Text", "Mode": "Constant", "Role": "Content" },
                { "Name": "TASK_WORK_PATH", "Type": "Text", "Mode": "Constant" }
              ],
              "Bindings": { "正文": "校对@定稿", "TASK_WORK_PATH": "@TASK_WORK_PATH" }
            }
          ]
        }
      ]
    }
  ]
}
```

## 双版比选

同一容器「大纲稿」引用两次，各写一版大纲，再由执行节点比选大纲，按选定大纲写定稿，定稿写入本次任务目录的 `比选定稿.md`。

目标与风格同样是常量开始端口。大纲稿被引用两次，每个实例各自执行一次并产出自己的大纲常量。比选大纲静态读取两份常量，只交回选定的大纲原文，选择理由放在回复里。撰写从选定大纲一次写出定稿，中间不复制全文，避免在模型端口之间搬运长文本。

    {

      ],
      "Outputs": [{ "Name": "大纲", "Type": "Text", "Mode": "Constant" }]
    },
    {
      "Name": "大纲稿",

          { "Name": "TASK_WORK_PATH", "Type": "Text", "Mode": "Constant", "Description": "本次任务目录。" },
          { "Name": "MODEL_NAME", "Type": "Text", "Mode": "Constant" }
        ]
      },
      "HostStartInputs": {
        "TASK_WORK_PATH": "TASK_WORK_PATH",
        "MODEL_NAME": "MODEL"
      },
      "Nodes": [
        {
          "Name": "整体",
          "Kind": "Container",
          "Inputs": [
            { "Name": "目标", "Type": "Text", "Mode": "Constant", "Role": "Content" },
