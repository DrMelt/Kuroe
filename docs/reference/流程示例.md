# 流程示例

示例按文件分块，节点库写入 `.kuroe/nodes.json`，流程写入 `.kuroe/flows.json`，哪些开始端口由宿主给值写入 `.kuroe/hostStartInputs.json`。两个示例都通过导入校验。字段说明见[流程定义](流程定义.md)，节点类型见[节点](nodes/index.md)。

## 单线撰写

拟定大纲，经容器写出全文，再校对，定稿写入本次任务目录的 `单线定稿.md`。

目标与风格在提交时给一次，`TASK_WORK_PATH` 与 `MODEL_NAME` 两个开始端口由宿主给值。容器「成稿」只包含撰写成员，直接对外输出正文。

宿主给值由 `.kuroe/hostStartInputs.json` 声明，键是流程名，值是开始端口到宿主输入名的映射：

```json
{
  "HostStartInputs": {
    "单线撰写": { "TASK_WORK_PATH": "TASK_WORK_PATH", "MODEL_NAME": "MODEL" },
    "双稿比选": { "MODEL_NAME": "MODEL" }
  }
}
```

节点库写入 `.kuroe/nodes.json`：

```json
{
  "Nodes": [
    {
      "Name": "撰写正文",
      "Kind": "Llm",
      "Emission": "Once",
      "Parameters": { "Prompt": "按大纲撰写正文。" },
      "Inputs": [{ "Name": "大纲", "Type": "Text", "Mode": "Constant", "Role": "Content" }],
      "Outputs": [{ "Name": "正文", "Type": "Text", "Mode": "Constant" }]
    },
    {
      "Name": "拟定大纲",
      "Kind": "Llm",
      "Emission": "Once",
      "Parameters": { "Prompt": "写出可撰写的章节大纲。" },
      "Inputs": [
        { "Name": "目标", "Type": "Text", "Mode": "Constant", "Role": "Content" },
        { "Name": "风格", "Type": "Text", "Mode": "Constant", "Role": "Content" }
      ],
      "Outputs": [{ "Name": "大纲", "Type": "Text", "Mode": "Constant" }]
    },
    {
      "Name": "校对全文",
      "Kind": "Llm",
      "Emission": "Once",
      "Parameters": { "Prompt": "校对全文，交回修正后的定稿。" },
      "Inputs": [{ "Name": "全文", "Type": "Text", "Mode": "Constant", "Role": "Content" }],
      "Outputs": [{ "Name": "定稿", "Type": "Text", "Mode": "Constant" }]
    },
    {
      "Name": "成稿",
      "Kind": "Container",
      "Inputs": [
        { "Name": "大纲", "Type": "Text", "Mode": "Constant", "Role": "Content" },
        { "Name": "MODEL_NAME", "Type": "Text", "Mode": "Constant" }
      ],
      "Outputs": [{ "Name": "全文", "Type": "Text", "Mode": "Constant", "Ref": "撰写@正文" }],
      "Nodes": [
        { "Label": "撰写", "Use": "撰写正文", "Bindings": { "大纲": "@大纲", "MODEL_NAME": "@MODEL_NAME" } }
      ]
    }
  ]
}
```

流程写入 `.kuroe/flows.json`：

```json
{
  "Flows": [
    {
      "Name": "单线撰写",
      "Description": "拟定大纲后经容器写出全文，再校对并写入本次任务目录",
      "Ports": [
        { "Name": "目标", "Type": "Text", "Mode": "Constant" },
        { "Name": "风格", "Type": "Text", "Mode": "Constant" },
        { "Name": "TASK_WORK_PATH", "Type": "Text", "Mode": "Constant" },
        { "Name": "MODEL_NAME", "Type": "Text", "Mode": "Constant" }
      ],
      "Root": {
        "Name": "整体",
        "Kind": "Container",
        "Inputs": [
          { "Name": "目标", "Type": "Text", "Mode": "Constant", "Role": "Content" },
          { "Name": "风格", "Type": "Text", "Mode": "Constant", "Role": "Content" },
          { "Name": "TASK_WORK_PATH", "Type": "Text", "Mode": "Constant" },
          { "Name": "MODEL_NAME", "Type": "Text", "Mode": "Constant" }
        ],
        "Bindings": {
          "目标": "@目标",
          "风格": "@风格",
          "TASK_WORK_PATH": "@TASK_WORK_PATH",
          "MODEL_NAME": "@MODEL_NAME"
        },
        "Nodes": [
          { "Label": "拟定", "Use": "拟定大纲", "Bindings": { "目标": "@目标", "风格": "@风格", "MODEL_NAME": "@MODEL_NAME" } },
          { "Label": "成稿", "Use": "成稿", "Bindings": { "大纲": "拟定@大纲", "MODEL_NAME": "@MODEL_NAME" } },
          { "Label": "校对", "Use": "校对全文", "Bindings": { "全文": "成稿@全文", "MODEL_NAME": "@MODEL_NAME" } },
          {
            "Name": "保存定稿",
            "Kind": "FileWrite",
            "Parameters": { "Path": "{root}/单线定稿.md", "Variables": { "root": "TASK_WORK_PATH" } },
            "Inputs": [
              { "Name": "正文", "Type": "Text", "Mode": "Constant", "Role": "Content" },
              { "Name": "TASK_WORK_PATH", "Type": "Text", "Mode": "Constant" }
            ],
            "Bindings": { "正文": "校对@定稿", "TASK_WORK_PATH": "@TASK_WORK_PATH" }
          }
        ]
      }
    }
  ]
}
```

## 容器引用两次

同一容器定义引用两次，每个实例各自执行一次，产出各自的大纲常量。比选节点静态读取两份常量，只交回选定的大纲原文，选择理由放在回复里。

节点库写入 `.kuroe/nodes.json`：

```json
{
  "Nodes": [
    {
      "Name": "拟定大纲",
      "Kind": "Llm",
      "Emission": "Once",
      "Parameters": { "Prompt": "写出可撰写的章节大纲。" },
      "Inputs": [
        { "Name": "目标", "Type": "Text", "Mode": "Constant", "Role": "Content" },
        { "Name": "风格", "Type": "Text", "Mode": "Constant", "Role": "Content" }
      ],
      "Outputs": [{ "Name": "大纲", "Type": "Text", "Mode": "Constant" }]
    },
    {
      "Name": "撰写正文",
      "Kind": "Llm",
      "Emission": "Once",
      "Parameters": { "Prompt": "按大纲撰写正文。" },
      "Inputs": [{ "Name": "大纲", "Type": "Text", "Mode": "Constant", "Role": "Content" }],
      "Outputs": [{ "Name": "正文", "Type": "Text", "Mode": "Constant" }]
    },
    {
      "Name": "大纲稿",
      "Kind": "Container",
      "Inputs": [
        { "Name": "目标", "Type": "Text", "Mode": "Constant", "Role": "Content" },
        { "Name": "风格", "Type": "Text", "Mode": "Constant", "Role": "Content" },
        { "Name": "MODEL_NAME", "Type": "Text", "Mode": "Constant" }
      ],
      "Outputs": [{ "Name": "大纲", "Type": "Text", "Mode": "Constant", "Ref": "拟定@大纲" }],
      "Nodes": [
        { "Label": "拟定", "Use": "拟定大纲", "Bindings": { "目标": "@目标", "风格": "@风格", "MODEL_NAME": "@MODEL_NAME" } }
      ]
    }
  ]
}
```

流程写入 `.kuroe/flows.json`：

```json
{
  "Flows": [
    {
      "Name": "双稿比选",
      "Description": "同一容器引用两次，比选两份大纲后撰写正文",
      "Ports": [
        { "Name": "目标", "Type": "Text", "Mode": "Constant" },
        { "Name": "风格", "Type": "Text", "Mode": "Constant" },
        { "Name": "MODEL_NAME", "Type": "Text", "Mode": "Constant" }
      ],
      "Root": {
        "Name": "整体",
        "Kind": "Container",
        "Inputs": [
          { "Name": "目标", "Type": "Text", "Mode": "Constant", "Role": "Content" },
          { "Name": "风格", "Type": "Text", "Mode": "Constant", "Role": "Content" },
          { "Name": "MODEL_NAME", "Type": "Text", "Mode": "Constant" }
        ],
        "Bindings": { "目标": "@目标", "风格": "@风格", "MODEL_NAME": "@MODEL_NAME" },
        "Nodes": [
          { "Label": "甲纲", "Use": "大纲稿", "Bindings": { "目标": "@目标", "风格": "@风格", "MODEL_NAME": "@MODEL_NAME" } },
          { "Label": "乙纲", "Use": "大纲稿", "Bindings": { "目标": "@目标", "风格": "@风格", "MODEL_NAME": "@MODEL_NAME" } },
          {
            "Name": "比选大纲",
            "Kind": "Llm",
            "Emission": "Once",
            "Bindings": { "甲纲": "甲纲@大纲", "乙纲": "乙纲@大纲", "MODEL_NAME": "@MODEL_NAME" },
            "Parameters": { "Prompt": "比较两版大纲，回复里给出选择理由；端口只交回选定的一版大纲原文。" },
            "Inputs": [
              { "Name": "甲纲", "Type": "Text", "Mode": "Constant", "Role": "Content" },
              { "Name": "乙纲", "Type": "Text", "Mode": "Constant", "Role": "Content" }
            ],
            "Outputs": [{ "Name": "选定大纲", "Type": "Text", "Mode": "Constant" }]
          },
          { "Label": "撰写", "Use": "撰写正文", "Bindings": { "大纲": "比选大纲@选定大纲", "MODEL_NAME": "@MODEL_NAME" } }
        ]
      }
    }
  ]
}
```
