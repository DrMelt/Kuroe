# 流程示例

## 单线撰写

制定大纲后引用容器「成稿」写出全文，再引用库节点「校对全文」校对，定稿写入工作区 `out/单线定稿.md`。

目标与风格在提交时给一次，声明为常量开始端口。制定大纲声明 `Emission: Once`，输入全部是常量，静态输入就绪后执行并首次发布后冻结；后续节点静态读取上一步的常量产出，按依赖依次执行。

```json
{
  "Nodes": [
    {
      "Name": "撰写正文",
      "Emission": "Once",
      "Kind": "Llm",
      "Prompt": "按大纲撰写正文。",
      "Tools": ["file"],
      "ToolPorts": { "file": { "root": "TASK_WORK_PATH" } },
      "Inputs": [
        { "Name": "大纲", "Type": "Text", "Mode": "Constant", "Role": "Content", "Description": "待撰写的章节大纲。" },
        { "Name": "TASK_WORK_PATH", "Type": "Text", "Mode": "Constant", "Description": "本次任务目录，文件工具的路径基准。" }
      ],
      "Outputs": [{ "Name": "正文", "Type": "Text", "Mode": "Constant", "Description": "按大纲写好的章节正文。" }]
    },
    {
      "Name": "合成全文",
      "Emission": "Once",
      "Kind": "Llm",
      "Prompt": "把正文合成一篇完整文档。",
      "Inputs": [{ "Name": "正文", "Type": "Text", "Mode": "Constant", "Role": "Content", "Description": "章节正文。" }],
      "Outputs": [{ "Name": "全文", "Type": "Text", "Mode": "Constant" }]
    },
    {
      "Name": "校对全文",
      "Kind": "Llm",
      "Emission": "Once",
      "Prompt": "校对全文，交回修正后的定稿。",
      "Inputs": [{ "Name": "全文", "Type": "Text", "Mode": "Constant", "Role": "Content", "Description": "待校对的全文。" }],
      "Outputs": [{ "Name": "定稿", "Type": "Text", "Mode": "Constant" }]
    },
    {
      "Name": "成稿",
      "Kind": "Container",
      "Inputs": [
        { "Name": "大纲", "Type": "Text", "Mode": "Constant", "Role": "Content", "Description": "待撰写的章节大纲。" },
        { "Name": "TASK_WORK_PATH", "Type": "Text", "Mode": "Constant", "Description": "本次任务目录。" },
        { "Name": "MODEL_NAME", "Type": "Text", "Mode": "Constant", "Description": "执行使用的模型，由宿主给出。" }
      ],
      "Outputs": [{ "Name": "全文", "Type": "Text", "Mode": "Constant", "Ref": "合成@全文", "Description": "容器内写好的全文。" }],
      "Nodes": [
        { "Label": "撰写", "Use": "撰写正文", "Static": { "大纲": "@大纲", "MODEL_NAME": "@MODEL_NAME", "TASK_WORK_PATH": "@TASK_WORK_PATH" } },
        { "Label": "合成", "Use": "合成全文", "Static": { "正文": "撰写@正文", "MODEL_NAME": "@MODEL_NAME" } }
      ]
    }
  ],
  "Flows": [
    {
      "Name": "单线撰写",
      "Description": "制定大纲后经容器写出全文，再引用库节点校对，定稿写入文件",
      "Start": {
        "Outputs": [
          { "Name": "目标", "Type": "Text", "Mode": "Constant", "Description": "要写的文档主题。" },
          { "Name": "风格", "Type": "Text", "Mode": "Constant", "Description": "写作风格要求。" },
          { "Name": "WORKSPACE_PATH", "Type": "Text", "Mode": "Constant", "Description": "工作区根目录。" },
          { "Name": "TASK_WORK_PATH", "Type": "Text", "Mode": "Constant", "Description": "本次任务目录。" },
          { "Name": "MODEL_NAME", "Type": "Text", "Mode": "Constant" }
        ]
      },
      "HostStartInputs": {
        "WORKSPACE_PATH": "WORKSPACE_PATH",
        "TASK_WORK_PATH": "TASK_WORK_PATH",
        "MODEL_NAME": "MODEL"
      },
      "Nodes": [
        {
          "Name": "整体",
          "Kind": "Container",
          "Inputs": [
            { "Name": "目标", "Type": "Text", "Mode": "Constant", "Role": "Content" },
            { "Name": "风格", "Type": "Text", "Mode": "Constant", "Role": "Content" },
            { "Name": "WORKSPACE_PATH", "Type": "Text", "Mode": "Constant" },
            { "Name": "TASK_WORK_PATH", "Type": "Text", "Mode": "Constant" },
            { "Name": "MODEL_NAME", "Type": "Text", "Mode": "Constant" }
          ],
          "In": {
            "目标": "Start@目标",
            "风格": "Start@风格",
            "WORKSPACE_PATH": "Start@WORKSPACE_PATH",
            "TASK_WORK_PATH": "Start@TASK_WORK_PATH",
            "MODEL_NAME": "Start@MODEL_NAME"
          },
          "Nodes": [
            {
              "Name": "制定大纲",
              "Emission": "Once",
              "Kind": "Llm",
              "Static": { "目标": "@目标", "风格": "@风格", "MODEL_NAME": "@MODEL_NAME" },
              "Prompt": "写出可撰写的章节大纲。",
              "Inputs": [
                { "Name": "目标", "Type": "Text", "Mode": "Constant", "Role": "Content", "Description": "任务目标。" },
                { "Name": "风格", "Type": "Text", "Mode": "Constant", "Role": "Content", "Description": "写作风格要求。" }
              ],
              "Outputs": [{ "Name": "大纲", "Type": "Text", "Mode": "Constant" }]
            },
            {
              "Label": "成稿",
              "Use": "成稿",
              "In": {
                "大纲": "制定大纲@大纲",
                "TASK_WORK_PATH": "@TASK_WORK_PATH",
                "MODEL_NAME": "@MODEL_NAME"
              }
            },
            {
              "Label": "校对",
              "Use": "校对全文",
              "Static": { "全文": "成稿@全文", "MODEL_NAME": "@MODEL_NAME" }
            },
            {
              "Name": "保存定稿",
              "Kind": "FileWrite",
              "Path": "{root}/out/单线定稿.md",
              "Variables": { "root": "WORKSPACE_PATH" },
              "Inputs": [
                { "Name": "正文", "Type": "Text", "Mode": "Constant", "Role": "Content" },
                { "Name": "WORKSPACE_PATH", "Type": "Text", "Mode": "Constant" }
              ],
              "Static": { "正文": "校对@定稿", "WORKSPACE_PATH": "@WORKSPACE_PATH" }
            }
          ]
        }
      ]
    }
  ]
}
```

## 双版比选

同一容器「成稿」引用两次，各写一版全文，再由执行节点比选，选定的定稿写入工作区 `out/比选定稿.md`。

目标与风格同样是常量开始端口。成稿被引用两次，每个实例各自执行一次并产出自己的全文常量；比选静态读取两份常量，定稿写入文件。

```json
{
  "Nodes": [
    {
      "Name": "撰写正文",
      "Emission": "Once",
      "Kind": "Llm",
      "Prompt": "按大纲撰写正文。",
```
